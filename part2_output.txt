-- CSCI 341 Assignment 3 - Research Conference and Paper Review Management System
-- PostgreSQL schema. Re-runnable: drops existing objects first.

DROP VIEW IF EXISTS accepted_paper_schedule;
DROP TABLE IF EXISTS review, paper_version, paper_author, paper_keyword, paper,
    conference_session, track, conference_reviewer, conference_host, conference,
    reviewer_expertise, reviewer, author_research_area, author, app_user CASCADE;

-- Users and roles (disjoint=no / overlapping, total participation: author and/or reviewer)
CREATE TABLE app_user (
    user_id       INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    given_name    VARCHAR(100) NOT NULL,
    surname       VARCHAR(100) NOT NULL,
    email         VARCHAR(255) NOT NULL,
    affiliation   VARCHAR(200) NOT NULL,
    country       VARCHAR(100) NOT NULL,
    phone_number  VARCHAR(30)  NOT NULL,
    CONSTRAINT uq_app_user_email UNIQUE (email)
);

CREATE TABLE author (
    author_id INTEGER PRIMARY KEY
        REFERENCES app_user(user_id) ON DELETE CASCADE,
    orcid     VARCHAR(19),
    CONSTRAINT uq_author_orcid UNIQUE (orcid)
);

CREATE TABLE author_research_area (
    author_id  INTEGER NOT NULL REFERENCES author(author_id) ON DELETE CASCADE,
    area_label VARCHAR(100) NOT NULL,
    PRIMARY KEY (author_id, area_label)
);

CREATE TABLE reviewer (
    reviewer_id      INTEGER PRIMARY KEY
        REFERENCES app_user(user_id) ON DELETE CASCADE,
    years_experience INTEGER NOT NULL,
    CONSTRAINT ck_reviewer_years CHECK (years_experience >= 0)
);

CREATE TABLE reviewer_expertise (
    reviewer_id INTEGER NOT NULL REFERENCES reviewer(reviewer_id) ON DELETE CASCADE,
    area_label  VARCHAR(100) NOT NULL,
    PRIMARY KEY (reviewer_id, area_label)
);

-- Conferences
CREATE TABLE conference (
    conference_id       INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name                VARCHAR(200) NOT NULL,
    acronym             VARCHAR(30)  NOT NULL,
    description         TEXT         NOT NULL,
    city                VARCHAR(100) NOT NULL,
    country             VARCHAR(100) NOT NULL,
    start_date          DATE NOT NULL,
    end_date            DATE NOT NULL,
    submission_deadline DATE NOT NULL,
    review_deadline     DATE NOT NULL,
    CONSTRAINT ck_conference_dates CHECK (end_date >= start_date)
);

CREATE TABLE conference_host (
    conference_id INTEGER NOT NULL REFERENCES conference(conference_id) ON DELETE CASCADE,
    host_name     VARCHAR(200) NOT NULL,
    PRIMARY KEY (conference_id, host_name)
);

-- Reviewer pool (M:N). Removing a conference removes only the pool memberships.
CREATE TABLE conference_reviewer (
    conference_id INTEGER NOT NULL REFERENCES conference(conference_id) ON DELETE CASCADE,
    reviewer_id   INTEGER NOT NULL REFERENCES reviewer(reviewer_id) ON DELETE CASCADE,
    PRIMARY KEY (conference_id, reviewer_id)
);

CREATE TABLE track (
    track_id       INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    conference_id  INTEGER NOT NULL REFERENCES conference(conference_id) ON DELETE CASCADE,
    name           VARCHAR(200) NOT NULL,
    description    TEXT         NOT NULL,
    research_area  VARCHAR(100) NOT NULL,
    coordinator_id INTEGER NOT NULL REFERENCES reviewer(reviewer_id) ON DELETE RESTRICT
);

CREATE TABLE conference_session (
    session_id    INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    conference_id INTEGER NOT NULL REFERENCES conference(conference_id) ON DELETE CASCADE,
    title         VARCHAR(200) NOT NULL,
    session_date  DATE NOT NULL,
    start_time    TIME NOT NULL,
    end_time      TIME NOT NULL,
    room_label    VARCHAR(100) NOT NULL,
    chair_id      INTEGER NOT NULL REFERENCES reviewer(reviewer_id) ON DELETE RESTRICT,
    CONSTRAINT ck_session_times CHECK (end_time > start_time)
);

-- Papers. session_id is NULL when the paper is not scheduled (at most one session per paper).
CREATE TABLE paper (
    paper_id        INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    track_id        INTEGER NOT NULL REFERENCES track(track_id) ON DELETE RESTRICT,
    title           VARCHAR(300) NOT NULL,
    abstract        TEXT NOT NULL,
    submission_date DATE NOT NULL,
    status          VARCHAR(20) NOT NULL DEFAULT 'submitted',
    session_id      INTEGER REFERENCES conference_session(session_id) ON DELETE SET NULL,
    CONSTRAINT ck_paper_status CHECK (status IN
        ('submitted', 'under review', 'accepted', 'rejected', 'withdrawn'))
);

CREATE TABLE paper_keyword (
    paper_id INTEGER NOT NULL REFERENCES paper(paper_id) ON DELETE CASCADE,
    keyword  VARCHAR(100) NOT NULL,
    PRIMARY KEY (paper_id, keyword)
);

CREATE TABLE paper_author (
    paper_id         INTEGER NOT NULL REFERENCES paper(paper_id) ON DELETE CASCADE,
    author_id        INTEGER NOT NULL REFERENCES author(author_id) ON DELETE RESTRICT,
    author_position  INTEGER NOT NULL,
    is_corresponding BOOLEAN NOT NULL DEFAULT FALSE,
    PRIMARY KEY (paper_id, author_id),
    CONSTRAINT uq_paper_author_position UNIQUE (paper_id, author_position),
    CONSTRAINT ck_paper_author_position CHECK (author_position > 0)
);

-- Weak entity: identified by (paper_id, version_no); removed with its paper.
CREATE TABLE paper_version (
    paper_id      INTEGER NOT NULL REFERENCES paper(paper_id) ON DELETE CASCADE,
    version_no    INTEGER NOT NULL,
    file_location VARCHAR(500) NOT NULL,
    uploaded_at   TIMESTAMP NOT NULL,
    change_note   TEXT,
    PRIMARY KEY (paper_id, version_no),
    CONSTRAINT ck_paper_version_no CHECK (version_no > 0)
);

-- Completed reviews only: every review field except confidential_comments is mandatory.
CREATE TABLE review (
    review_id              INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    paper_id               INTEGER NOT NULL REFERENCES paper(paper_id) ON DELETE CASCADE,
    reviewer_id            INTEGER NOT NULL REFERENCES reviewer(reviewer_id) ON DELETE RESTRICT,
    overall_score          INTEGER NOT NULL,
    recommendation         VARCHAR(10) NOT NULL,
    comments_for_authors   TEXT NOT NULL,
    confidential_comments  TEXT,
    review_submission_date DATE NOT NULL,
    CONSTRAINT uq_review_paper_reviewer UNIQUE (paper_id, reviewer_id),
    CONSTRAINT ck_review_score CHECK (overall_score BETWEEN 1 AND 5),
    CONSTRAINT ck_review_recommendation CHECK (recommendation IN ('accept', 'reject'))
);

Task 6.3: Papers, accepted papers and average review score per conference
====================================================================================================

-- 6.3
SQL:
    WITH paper_stats AS (
        SELECT t.conference_id,
               COUNT(*)                                     AS total_papers,
               COUNT(*) FILTER (WHERE p.status = 'accepted') AS accepted_papers
        FROM paper p
        JOIN track t ON t.track_id = p.track_id
        GROUP BY t.conference_id
    ), review_stats AS (
        SELECT t.conference_id, AVG(rv.overall_score) AS avg_score
        FROM review rv
        JOIN paper p ON p.paper_id = rv.paper_id
        JOIN track t ON t.track_id = p.track_id
        GROUP BY t.conference_id
    )
    SELECT c.conference_id, c.acronym,
           COALESCE(ps.total_papers, 0)    AS total_papers,
           COALESCE(ps.accepted_papers, 0) AS accepted_papers,
           ROUND(rs.avg_score, 2)          AS avg_review_score
    FROM conference c
    LEFT JOIN paper_stats ps  ON ps.conference_id = c.conference_id
    LEFT JOIN review_stats rs ON rs.conference_id = c.conference_id
    ORDER BY c.conference_id

Task 7: Acceptance rate per conference (derived attribute)
====================================================================================================

-- 7
SQL:
    SELECT c.conference_id, c.acronym,
           COUNT(p.paper_id) FILTER (WHERE p.status = 'accepted') AS accepted_count,
           COUNT(p.paper_id) FILTER (WHERE p.status = 'rejected') AS rejected_count,
           ROUND(100.0 * COUNT(p.paper_id) FILTER (WHERE p.status = 'accepted')
                 / NULLIF(COUNT(p.paper_id) FILTER (WHERE p.status IN ('accepted', 'rejected')), 0),
                 2) AS acceptance_rate
    FROM conference c
    LEFT JOIN track t ON t.conference_id = c.conference_id
    LEFT JOIN paper p ON p.track_id = t.track_id
    GROUP BY c.conference_id, c.acronym
    ORDER BY c.conference_id

SQL:
    CREATE OR REPLACE VIEW accepted_paper_schedule AS
    SELECT p.paper_id,
           p.title                                   AS paper_title,
           c.conference_id,
           c.acronym                                 AS conference_acronym,
           t.name                                    AS track_name,
           s.session_id,
           s.title                                   AS session_title,
           s.session_date,
           s.start_time,
           s.end_time,
           s.room_label,
           ch.given_name || ' ' || ch.surname        AS chair_full_name,
           ca.given_name || ' ' || ca.surname        AS corresponding_author_name,
           ca.email                                  AS corresponding_author_email
    FROM paper p
    JOIN conference_session s ON s.session_id = p.session_id
    JOIN track t              ON t.track_id = p.track_id
    JOIN conference c         ON c.conference_id = s.conference_id
    JOIN app_user ch          ON ch.user_id = s.chair_id
    LEFT JOIN paper_author pa ON pa.paper_id = p.paper_id AND pa.is_corresponding
    LEFT JOIN app_user ca     ON ca.user_id = pa.author_id;
